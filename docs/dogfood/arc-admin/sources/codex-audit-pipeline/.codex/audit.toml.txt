# ============================================================
# 硬性约束：SQL 写操作只允许出现在 Repository / 迁移 / 测试层
# 这是与"模板生成代码"兼容的关键：Repository 模板里本来就有
# INSERT / UPDATE / .execute()，所以这里用 allowlist 放行写操作的家。
# ============================================================
[[hard_rules]]
name = "SQL 写操作仅允许出现在 Repository/迁移/测试层"
severity = "blocker"
paths = ["backend"]
extensions = ["rs", "sql"]
patterns = [
  '(?i)INSERT\s+INTO',
  '(?i)UPDATE\s+(?:"[^"]+"|[A-Za-z_][A-Za-z0-9_.]*)\s+SET',
  '(?i)DELETE\s+FROM',
  '(?i)DROP\s+TABLE',
  '(?i)ALTER\s+TABLE',
  '(?i)TRUNCATE',
  '(?:sqlx\s*::\s*)?raw_sql\s*\(',
  '(?:sqlx\s*::\s*)?QueryBuilder(?:\s*::<[^>]+>)?\s*::\s*new\s*\(',
  '\.execute\s*\(',
  '\.exec\s*\(',
]
allowlist = [
  { kind = "path-prefix", path = "backend/src/repositories" },
  { kind = "path-prefix", path = "backend/src/db.rs" },
  { kind = "path-prefix", path = "backend/migrations" },
  { kind = "path-prefix", path = "backend/tests" },
  { kind = "path-prefix", path = "backend/src/seed" },
]
exclude_patterns = []
